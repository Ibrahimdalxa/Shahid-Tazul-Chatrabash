Supabase Cloud Connection

Setup database tables

Your project connects to Supabase URL: https://ujkrxcqvnxkxfnpwaydy.supabase.co

To enable live cloud sync for all members and the cook across multiple phones, run this SQL script once in your Supabase SQL Editor: 

-- Creates mess_members, meal_entries & cooking_status
CREATE TABLE IF NOT EXISTS public.mess_members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'member',
    pin TEXT NOT NULL DEFAULT '0000',
    room_no TEXT DEFAULT '',
    avatar_color TEXT DEFAULT '#3B82F6',
    created_at TIMESTAMPTZ DEFAULT now()
);
CREATE TABLE IF NOT EXISTS public.meal_entries (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    member_id UUID NOT NULL REFERENCES public.mess_members(id) ON DELETE CASCADE,
    date DATE NOT NULL,
    meal_type TEXT NOT NULL,
    count INTEGER NOT NULL DEFAULT 1,
    guest_count INTEGER NOT NULL DEFAULT 0,
    CONSTRAINT unique_member_date_meal UNIQUE (member_id, date, meal_type)
);
CREATE TABLE IF NOT EXISTS public.cooking_status (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    date DATE NOT NULL,
    meal_type TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'pending',
    cooked_meals_count INTEGER NOT NULL DEFAULT 0,
    menu_item TEXT DEFAULT '',
    completed_at TIMESTAMPTZ,
    CONSTRAINT unique_date_meal_status UNIQUE (date, meal_type)
);
ALTER TABLE public.mess_members ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.meal_entries ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.cooking_status ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow anon all on mess_members" ON public.mess_members FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon all on meal_entries" ON public.meal_entries FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon all on cooking_status" ON public.cooking_status FOR ALL TO anon USING (true) WITH CHECK (true);
Step 1: Click Copy SQL above.
Step 2: Open your Supabase SQL Editor and paste it.
Step 3: Click Run in Supabase, then click the test button below!
